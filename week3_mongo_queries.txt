use('student_success');

// 1. Profile for StudentID 1 and all of their advising notes.
db.students.find(
  { "studentId": 1 },
  {
    "_id": 0,
    "studentId": 1,
    "firstName": 1,
    "lastName": 1,
    "email": 1,
    "program": 1,
    "major": 1,
    "advisingNotes": 1
  }
);

// 2. Finds support visits for StudentID 2 from the last 30 days.
db.students.aggregate([
  { $match: { "studentId": 2 } },
  {
    $project: {
      "_id": 0,
      "studentId": 1,
      "firstName": 1,
      "lastName": 1,
      "supportVisits": {
        "$filter": {
          "input": "$supportVisits",
          "as": "visit",
          "cond": {
            "$and": [
              { "$gte": ["$$visit.supportDate", "2026-08-28"] },
              { "$lte": ["$$visit.supportDate", "2026-09-26"] }
            ]
          }
        }
      }
    }
  }
]);

// 3. Students with 3 or more support visits are retrieved.
db.students.find(
  {
    "$expr": {
      "$gte": [
        { "$size": { "$ifNull": ["$supportVisits", []] } },
        3
      ]
    }
  },
  {
    "_id": 0,
    "studentId": 1,
    "firstName": 1,
    "lastName": 1,
    "supportVisits": 1
  }
);
